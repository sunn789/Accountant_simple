// src/App.tsx
import { BrowserRouter, Routes, Route } from 'react-router-dom';
import Navbar from './components/Navbar';
import Dashboard from './pages/Dashboard';
import InvoiceList from './pages/InvoiceList';
import InvoiceDetail from './pages/InvoiceDetail';
import CreateInvoice from './pages/CreateInvoice';
import './App.css';

function App() {
  return (
    <BrowserRouter>
      <Navbar />
      <div className="main-content">
        <Routes>
          <Route path="/" element={<Dashboard />} />
          <Route path="/invoices" element={<InvoiceList />} />
          <Route path="/invoices/new" element={<CreateInvoice />} />
          <Route path="/invoices/:id" element={<InvoiceDetail />} />
        </Routes>
      </div>
    </BrowserRouter>
  );
}

export default App;

  totalCount: number;
  pageNumber: number;
  pageSize: number;
  items: Invoice[];
}

  date: string;
  jalaliDate: string;
  invoiceCount: number;
  totalAmount: number;
}

  const response = await api.get('/Invoices/today-summary');
  return response.data;
};

export const getInvoices = async (
  page: number = 1,
  pageSize: number = 10,
  startDate?: string,
  endDate?: string,
  buyerName?: string,
  minAmount?: number,
  maxAmount?: number
): Promise<PagedResult> => {
  const params: Record<string, string | number> = {
    page,
    pageSize,
  };

  if (startDate) params.startDate = startDate;
  if (endDate) params.endDate = endDate;
  if (buyerName) params.buyerName = buyerName;
  if (minAmount !== undefined) params.minAmount = minAmount;
  if (maxAmount !== undefined) params.maxAmount = maxAmount;

  const response = await api.get('/Invoices', { params });

  return response.data;
};

import type { Invoice, InvoiceItem, Party } from '../../types/invoice';
import { saveInvoice } from '../../services/api';

const InvoiceForm = () => {
  const navigate = useNavigate();
  const [isSubmitting, setIsSubmitting] = useState(false);

  // ۱. استیت اطلاعات پایه
  const [invoiceNumber, setInvoiceNumber] = useState('');
 const [date, setDate] = useState(new Date().toISOString().split('T')[0]);
  const [paymentType, setPaymentType] = useState('cash');
  const [notes, setNotes] = useState('');

  // ۲. استیت اطلاعات خریدار
  const [buyer, setBuyer] = useState<Party>({
    name: '', economicCode: '', registrationNumber: '', nationalId: '',
    province: '', city: '', address: '', postalCode: '', phone: '', mobile: ''
  });

  // ۳. استیت اقلام فاکتور
  const [items, setItems] = useState<InvoiceItem[]>([
    { description: '', quantity: 1, unitPrice: 0, discount: 0, tax: 0 }
  ]);

  // هندلر تغییر فیلدهای خریدار
  const handleBuyerChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const { name, value } = e.target;
    setBuyer(prev => ({ ...prev, [name]: value }));
  };

  // هندلر تغییر فیلدهای آیتم‌ها
  const handleItemChange = (index: number, field: keyof InvoiceItem, value: string | number) => {
    const newItems = [...items];
    newItems[index] = { ...newItems[index], [field]: value };
    setItems(newItems);
  };

  // افزودن ردیف جدید
  const addItem = () => {
    setItems([...items, { description: '', quantity: 1, unitPrice: 0, discount: 0, tax: 0 }]);
  };

  // حذف ردیف
  const removeItem = (index: number) => {
    if (items.length > 1) {
      setItems(items.filter((_, i) => i !== index));
    }
  };

  // محاسبه جمع کل یک ردیف
  const calculateRowTotal = (item: InvoiceItem) => {
    return (item.quantity * item.unitPrice) - item.discount + item.tax;
  };

  // محاسبه جمع کل فاکتور
  const calculateGrandTotal = () => {
    return items.reduce((total, item) => total + calculateRowTotal(item), 0);
  };

  // ارسال به بک‌اند
  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setIsSubmitting(true);

    // فروشنده فیک (در نسخه واقعی باید از تنظیمات یا دیتابیس خوانده شود)
    const seller: Party = {
      name: 'شرکت من', economicCode: '123456789', registrationNumber: '111',
      nationalId: '1234567890', province: 'تهران', city: 'تهران',
      address: 'آدرس شرکت', postalCode: '1234567890', phone: '021-00000000', mobile: '09120000000'
    };

    const newInvoice: Invoice = {
      invoiceNumber: invoiceNumber || `INV-${Date.now()}`,
      date,
      seller,
      buyer,
      items: items.map(item => ({ ...item, totalPrice: calculateRowTotal(item) })),
      totalAmount: calculateGrandTotal(),
      paymentType,
      notes
    };

    try {
      await saveInvoice(newInvoice);
      alert('فاکتور با موفقیت ذخیره شد!');
      navigate('/invoices'); // هدایت به لیست فاکتورها
    } catch (error) {
      console.error(error);
      alert('خطا در ذخیره فاکتور. لطفا دوباره تلاش کنید.');
    } finally {
      setIsSubmitting(false);
    }
  };

  return (
    <form className="invoice-form" onSubmit={handleSubmit} dir="rtl">
      
      {/* بخش اطلاعات پایه */}

        </div>
        <div className="form-group">
          <label>تاریخ</label>
          <input type="text" value={date} onChange={(e) => setDate(e.target.value)} />
        </div>
      </div>

      {/* بخش اطلاعات خریدار */}
      <fieldset className="form-section">
        <legend>مشخصات خریدار</legend>
        <div className="form-row">
          <div className="form-group">
            <label>نام خریدار *</label>

          </div>
          <div className="form-group">
            <label>شماره اقتصادی</label>
            <input type="text" name="economicCode" value={buyer.economicCode} onChange={handleBuyerChange} />
          </div>
          <div className="form-group">

            <input type="text" name="nationalId" value={buyer.nationalId} onChange={handleBuyerChange} />
          </div>
        </div>
        <div className="form-row">
          <div className="form-group">
            <label>استان</label>
            <input type="text" name="province" value={buyer.province} onChange={handleBuyerChange} />
          </div>
          <div className="form-group">

// تبدیل تاریخ میلادی API به شمسی، بدون جابه‌جایی روز بر اثر timezone
const formatToJalali = (dateString: string) => {
  if (!dateString) return '-';

  try {
    const [datePart] = dateString.split('T');
    const [year, month, day] = datePart.split('-').map(Number);

    if (!year || !month || !day) return '-';

    const date = new Date(Date.UTC(year, month - 1, day));

    return new Intl.DateTimeFormat('fa-IR-u-ca-persian', {
      year: 'numeric',
      month: '2-digit',
      day: '2-digit',
      timeZone: 'UTC',
    }).format(date);
  } catch {
    return '-';
  }
};

// تبدیل تاریخ شمسی انتخاب‌شده به میلادی برای API
const convertToApiDate = (date: DateObject | null) => {
  if (!date) return undefined;

  const gregorianDate = date.convert(gregorian);

  return gregorianDate
    .format('YYYY-MM-DD')
    .replace(/[۰-۹]/g, (digit) =>
      String('۰۱۲۳۴۵۶۷۸۹'.indexOf(digit))
    )
    .replace(/[٠-٩]/g, (digit) =>
      String('٠١٢٣٤٥٦٧٨٩'.indexOf(digit))
    );
};

const formatToJalali = (dateString: string) => {
  if (!dateString) return '-';

  try {
    const datePart = dateString.split('T')[0];
    const [year, month, day] = datePart.split('-').map(Number);

    if (!year || !month || !day) return '-';

    const date = new Date(Date.UTC(year, month - 1, day));

    return new Intl.DateTimeFormat('fa-IR-u-ca-persian', {
      year: 'numeric',
      month: '2-digit',
      day: '2-digit',
      timeZone: 'UTC',
    }).format(date);
  } catch {
    return '-';
  }
};

const InvoiceDetail = () => {

  const navigate = useNavigate();

  const [error, setError] = useState('');

  useEffect(() => {
    let isMounted = true;

    const loadInvoice = async () => {
      const invoiceId = Number(id);

      if (!id || !Number.isInteger(invoiceId) || invoiceId <= 0) {
        setError('شناسه فاکتور معتبر نیست.');
        setLoading(false);

        if (isMounted) {

      isMounted = false;
    };
  }, [id]);

            var totalCustomers = await _context.Invoices

        [HttpPost]
        public async Task<IActionResult> CreateInvoice(
    [FromBody] Invoice invoice,
    CancellationToken cancellationToken)
        {
            await using var transaction =
                await _context.Database.BeginTransactionAsync(
                    cancellationToken
                );

            try
            {
                if (invoice.Date == default)
                {
                    invoice.Date = TimeZoneInfo.ConvertTimeFromUtc(
                        DateTime.UtcNow,
                        GetTehranTimeZone()
                    );
                }

                // مدیریت فروشنده
                if (invoice.Seller != null && invoice.Seller.Id == 0)
                {
                    _context.Parties.Add(invoice.Seller);

                    await _context.SaveChangesAsync(cancellationToken);

                    invoice.SellerId = invoice.Seller.Id;
                }
                else if (invoice.Seller != null)
                {
                    invoice.SellerId = invoice.Seller.Id;
                }

                invoice.Seller = null;

                // مدیریت خریدار
                if (invoice.Buyer != null && invoice.Buyer.Id == 0)
                {
                    _context.Parties.Add(invoice.Buyer);

                    await _context.SaveChangesAsync(cancellationToken);

                    invoice.BuyerId = invoice.Buyer.Id;
                }
                else if (invoice.Buyer != null)
                {
                    invoice.BuyerId = invoice.Buyer.Id;
                }

                invoice.Buyer = null;

                // ثبت فاکتور
                _context.Invoices.Add(invoice);

                await _context.SaveChangesAsync(cancellationToken);

                // ثبت عملیات صدور در Outbox
                await _invoiceIssuingService.EnqueueAsync(
                    invoice.Id,
                    cancellationToken
                );

                // ذخیره پیام Outbox
                await _context.SaveChangesAsync(cancellationToken);

                // تأیید نهایی تمام عملیات
                await transaction.CommitAsync(cancellationToken);

                return Ok(new
                {
                    Message = "فاکتور با موفقیت ذخیره شد",
                    InvoiceId = invoice.Id
                });
            }
            catch (Exception ex)
            {
                await transaction.RollbackAsync(
                    CancellationToken.None
                );

                return BadRequest(new
                {
                    Message = "خطا در ذخیره‌سازی فاکتور",
                    Error = ex.Message,
                    InnerError = ex.InnerException?.Message
                });
            }
        }

    public PublicInvoicesController(
        AppDbContext context,
        InvoiceFileStorage storage)
    {
        _context = context;
        _storage = storage;
    }

    [HttpGet("{token}")]
    public async Task<IActionResult> ViewInvoice(
        string token,
        CancellationToken cancellationToken)
    {
        if (string.IsNullOrWhiteSpace(token) || token.Length != 64)
            return NotFound();

        var tokenHash = InvoiceIssuingService.HashToken(token);

        var document = await _context.InvoiceDocuments
            .AsNoTracking()
            .FirstOrDefaultAsync(
                x =>
                    x.AccessTokenHash == tokenHash &&
                    x.IsActive &&
                    (x.ExpiresAt == null ||
                     x.ExpiresAt > DateTime.UtcNow),
                cancellationToken
            );

        if (document is null)
            return NotFound("لینک فاکتور معتبر نیست یا منقضی شده است.");

        var file = await _storage.ReadAsync(
            document.FilePath,
            cancellationToken
        );

        if (file is null)
            return NotFound("فایل فاکتور پیدا نشد.");

        Response.Headers.CacheControl = "no-store";
        Response.Headers["X-Content-Type-Options"] = "nosniff";

        return File(
            file,
            "application/pdf",
            $"Invoice-{document.InvoiceId}.pdf"
        );
    }
}
